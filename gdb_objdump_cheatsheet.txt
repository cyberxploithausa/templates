------------------------
OBJDUMP
------------------------
Basic usage
objdump -d ./binary
objdump -D ./binary
objdump -t ./binary
objdump -T ./binary
objdump -h ./binary
objdump -x ./binary
objdump -s -j .rodata ./binary
objdump -M intel -d ./binary
objdump -M att -d ./binary
objdump -d -Mintel --no-show-raw-insn ./binary

Disassemble specific address/range
objdump -d --start-address=0x4005d0 --stop-address=0x400610 ./binary

Symbol / relocation
objdump -t ./binary | less
objdump -R ./binary
readelf -s ./binary

Hex dump / raw data
objdump -s -j .data ./bin
hexdump -C -n 256 -s 0x100 ./binary

Examples — direct
objdump -d ./vuln | sed -n '/<main>:/,/<_start>:/p'
objdump -M intel -d ./vuln | sed -n '/<main>:/,/<_start>:/p'
objdump -d ./vuln | awk '{print $2}' | grep -E '^[0-9a-f]'
objdump -h ./vuln
objdump -D ./bin | grep -n "call" | sed -n '1,120p'
objdump -x ./lib.so | grep SONAME -A2
objdump -t ./bin | grep FUNC

Advanced combos / pipelines
objdump -d ./vuln | awk '/^[[:xdigit:]]+:/ {print}' | sed 's/:\t/ /g' | cut -f1,2,3
objdump -d -M intel ./vuln > tmp.dis
gdb -q -ex "file ./vuln" -ex "set disassemble-next-line on" -ex "disassemble main" -ex "quit"

Useful flags quick
--disassemble-all (-D)
--disassemble-text (-d)
--no-show-raw-insn
-M intel|att

Cross references
addr=$(objdump -d ./vuln | grep "<main>:" -n -m1 | cut -d: -f1)

Short examples collection (one-liners)
objdump -d ./bin | sed -n '/<main>:/,/<_start>:/p' | less
objdump -D ./bin | grep -n "call" | sed -n '1,120p'
objdump -x ./lib.so | grep SONAME -A2
objdump -t ./bin | grep FUNC

Misc
objdump -p ./binary | grep NEEDED
objdump -t ./binary | grep __stack_chk_fail
objdump -R ./binary | grep GOT
